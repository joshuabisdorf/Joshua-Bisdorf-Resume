%-------------------------%
% Resume spacing variables
%
% Edit this file to tune vertical spacing across the resume.
% All values are LaTeX lengths. Use pt for fine-grained adjustments.
% Positive values add space. Negative values remove space.
%
% RME:
% Requires: This file is loaded before section formatting and custom resume commands.
% Modifies: Named LaTeX length registers that control resume vertical spacing.
% Effects: Centralizes spacing controls so layout can be adjusted from one file.
% Inputs: The length values assigned below.
% Outputs: Named spacing variables consumed by resume.tex and custom-commands.tex.

%-------------------------%
% Section heading spacing
%-------------------------%

% Space inserted above each top-level section heading, such as Education or Work Experience.
\newlength{\resumeSectionBeforeSpacing}
\setlength{\resumeSectionBeforeSpacing}{9pt}

% Space inserted below each top-level section rule before the section contents begin.
\newlength{\resumeSectionAfterSpacing}
\setlength{\resumeSectionAfterSpacing}{4pt}

%-------------------------%
% Primary heading spacing
%-------------------------%

% Space inserted before a primary heading, such as a school or company entry.
\newlength{\resumePrimaryHeadingBeforeSpacing}
\setlength{\resumePrimaryHeadingBeforeSpacing}{2pt}

% Space inserted after a primary heading before details or bullets begin.
\newlength{\resumePrimaryHeadingAfterSpacing}
\setlength{\resumePrimaryHeadingAfterSpacing}{-2pt}

%-------------------------%
% Secondary heading spacing
%-------------------------%

% Space inserted before a secondary heading, such as an additional role under the same company.
\newlength{\resumeSecondaryHeadingBeforeSpacing}
\setlength{\resumeSecondaryHeadingBeforeSpacing}{4pt}

% Space inserted after a secondary heading before bullets begin.
\newlength{\resumeSecondaryHeadingAfterSpacing}
\setlength{\resumeSecondaryHeadingAfterSpacing}{-2pt}

%-------------------------%
% Project heading spacing
%-------------------------%

% Space inserted before each project heading.
\newlength{\resumeProjectHeadingBeforeSpacing}
\setlength{\resumeProjectHeadingBeforeSpacing}{4pt}

% Space inserted after each project heading before bullets begin.
\newlength{\resumeProjectHeadingAfterSpacing}
\setlength{\resumeProjectHeadingAfterSpacing}{-2pt}

%-------------------------%
% Education detail spacing
%-------------------------%

% Space inserted before non-bulleted education detail rows, such as honors or coursework.
\newlength{\resumeEducationDetailBeforeSpacing}
\setlength{\resumeEducationDetailBeforeSpacing}{1pt}

% Space inserted after non-bulleted education detail rows.
\newlength{\resumeEducationDetailAfterSpacing}
\setlength{\resumeEducationDetailAfterSpacing}{-2pt}

%-------------------------%
% Nested item spacing
%-------------------------%

% Extra space inserted after nested sub-items created with \resumeSubItem.
\newlength{\resumeNestedSubItemAfterSpacing}
\setlength{\resumeNestedSubItemAfterSpacing}{-2pt}

%-------------------------%
% Heading list spacing
%-------------------------%

% Space above the first item in each heading list.
\newlength{\resumeHeadingListTopSep}
\setlength{\resumeHeadingListTopSep}{0pt}

% Space added for paragraph separation inside heading lists.
\newlength{\resumeHeadingListParsep}
\setlength{\resumeHeadingListParsep}{0pt}

% Space added before heading lists when they begin after a blank line or paragraph break.
\newlength{\resumeHeadingListPartopSep}
\setlength{\resumeHeadingListPartopSep}{0pt}

% Space between entries in heading lists. Keep this low because headings use explicit before/after spacing.
\newlength{\resumeHeadingListItemSep}
\setlength{\resumeHeadingListItemSep}{0pt}

% Space inserted after each heading list closes, usually at the end of a section.
\newlength{\resumeHeadingListAfterSpacing}
\setlength{\resumeHeadingListAfterSpacing}{-1pt}

%-------------------------%
% Bullet list spacing
%-------------------------%

% Space between a role/project heading and its first bullet.
\newlength{\resumeBulletListTopSep}
\setlength{\resumeBulletListTopSep}{3pt}

% Space added for paragraph separation inside bullet lists.
\newlength{\resumeBulletListParsep}
\setlength{\resumeBulletListParsep}{0pt}

% Space added before bullet lists when they begin after a blank line or paragraph break.
\newlength{\resumeBulletListPartopSep}
\setlength{\resumeBulletListPartopSep}{0pt}

% Space between bullets within the same role or project.
\newlength{\resumeBulletListItemSep}
\setlength{\resumeBulletListItemSep}{2pt}

% Space inserted after a bullet list closes before the next heading or section.
\newlength{\resumeBulletListAfterSpacing}
\setlength{\resumeBulletListAfterSpacing}{-2pt}
